# PhotoFlow — runtime dependencies (what a shipping build needs)
#
# Deliberately minimal: this is what gets installed into the build environment
# before running packaging/build.bat, and everything here ends up inside the
# installer. Installer size is a real conversion factor, so anything optional
# lives in requirements-extra.txt instead.
#
#     pip install -r requirements.txt
#
# What was moved OUT of this file, and why:
#   insightface + onnxruntime -> requirements-extra.txt (its pretrained weights
#       are licensed for non-commercial use only, and it adds hundreds of MB;
#       core/sface_backend.py provides an Apache-2.0 alternative that needs
#       nothing beyond OpenCV)
#   rembg                     -> requirements-extra.txt (large model runtime for
#       one optional effect; there's a built-in fallback)
#   openai + python-dotenv    -> requirements-extra.txt (cloud scene labelling is
#       opt-in and has a local fallback)

# --- Configuration ---------------------------------------------------------- #
PyYAML>=6.0,<7.0                     # utils/config.py

# --- Imaging ---------------------------------------------------------------- #
Pillow>=10.0,<13.0
numpy>=1.24,<3.0
opencv-python-headless>=4.8,<5.0     # blur/quality metrics, and the ONNX runtime
                                     # behind the SFace/YuNet face recognition
ImageHash>=4.3,<5.0                  # perceptual hashing (duplicate detection)

# --- Face detection --------------------------------------------------------- #
mediapipe>=0.10,<1.0                 # core feature, used by all three tools

# --- Desktop UI ------------------------------------------------------------- #
PyQt6>=6.5,<7.0

# --- Album export ----------------------------------------------------------- #
# Imported lazily (core/album/raster.py), only for layered PSD export, but kept
# here because that is a shipped feature rather than an optional extra.
psd-tools>=1.16,<2.0

# --- Clustering ------------------------------------------------------------- #
# Technically optional (core/person_cluster.py falls back to a greedy pass and
# core/album/slot_matcher.py to a simpler match), but agglomerative clustering
# measurably improves person grouping, which is a headline album feature.
scipy>=1.10,<2.0

# --- Authentication & licensing ---------------------------------------------- #
# core/auth.py talks to the production API (requests) and stores the refresh
# token in the OS credential store (keyring). Both were already required for
# production login/licensing to work at all, but were missing from this file --
# a release build venv created from just `pip install -r requirements.txt`
# would freeze without them, and ui_qt.main's licensing setup swallows the
# resulting ModuleNotFoundError silently (by design, so a licensing bug can't
# block startup), so the installed app would just quietly have no login/
# licensing at all. keyring's Windows Credential Manager backend pulls in
# pywin32-ctypes automatically on Windows; nothing extra to add here for that.
requests>=2.31,<3.0
keyring>=24.0,<26.0
